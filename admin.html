-- ============================================================ -- 05_vendedor_login.sql -- Permite que a Edge Function grave o vendedor apos criar a conta -- no Auth, e passa a trazer o e-mail na listagem do painel. -- Rode depois de 04_vendedores.sql -- ============================================================ -- ------------------------------------------------------------ -- Gravacao sem checagem de admin. -- Nao recebe grant para ninguem: so o service_role alcanca, -- e o service_role e usado apenas dentro da Edge Function, -- que por sua vez verifica is_admin() antes de chamar. -- ------------------------------------------------------------ create or replace function vendedor_salvar_interno( p_id bigint default null, p_nome text default null, p_telefone text default null, p_codigo text default null, p_comissao_pct numeric default null, p_ativo boolean default true, p_user_id uuid default null ) returns json language plpgsql security definer set search_path = public as $$ declare v_id bigint; v_cod text; begin if length(trim(coalesce(p_nome,''))) < 3 then raise exception 'Informe o nome do vendedor.'; end if; v_cod := lower(regexp_replace(coalesce(p_codigo,''), '[^a-zA-Z0-9]', '', 'g')); if length(v_cod) < 3 then raise exception 'O codigo precisa ter ao menos 3 letras ou numeros.'; end if; if p_comissao_pct is null or p_comissao_pct < 0 or p_comissao_pct > 100 then raise exception 'A comissao precisa ficar entre 0 e 100.'; end if; if exists (select 1 from vendedores where lower(codigo) = v_cod and (p_id is null or id <> p_id)) then raise exception 'Ja existe um vendedor com esse codigo.'; end if; if p_user_id is not null and exists ( select 1 from vendedores where user_id = p_user_id and (p_id is null or id <> p_id)) then raise exception 'Essa conta ja esta vinculada a outro vendedor.'; end if; if p_id is null then insert into vendedores (nome, telefone, codigo, comissao_pct, ativo, user_id) values (trim(p_nome), nullif(regexp_replace(coalesce(p_telefone,''),'\D','','g'),''), v_cod, p_comissao_pct, coalesce(p_ativo,true), p_user_id) returning id into v_id; else update vendedores set nome = trim(p_nome), telefone = nullif(regexp_replace(coalesce(p_telefone,''),'\D','','g'),''), codigo = v_cod, comissao_pct = p_comissao_pct, ativo = coalesce(p_ativo,true), -- so sobrescreve o vinculo quando um novo e informado user_id = coalesce(p_user_id, user_id) where id = p_id returning id into v_id; if v_id is null then raise exception 'Vendedor nao encontrado.'; end if; end if; return json_build_object('ok', true, 'id', v_id, 'codigo', v_cod); end $$; revoke all on function vendedor_salvar_interno(bigint, text, text, text, numeric, boolean, uuid) from anon, authenticated, public; -- ------------------------------------------------------------ -- A versao do painel agora so confere a permissao e delega. -- Continua funcionando para quem preferir colar o UUID na mao. -- ------------------------------------------------------------ create or replace function admin_vendedor_salvar( p_id bigint default null, p_nome text default null, p_telefone text default null, p_codigo text default null, p_comissao_pct numeric default null, p_ativo boolean default true, p_user_id uuid default null ) returns json language plpgsql security definer set search_path = public as $$ begin if not is_admin() then raise exception 'Acesso restrito.'; end if; return vendedor_salvar_interno(p_id, p_nome, p_telefone, p_codigo, p_comissao_pct, p_ativo, p_user_id); end $$; grant execute on function admin_vendedor_salvar(bigint, text, text, text, numeric, boolean, uuid) to authenticated; -- ------------------------------------------------------------ -- Listagem passa a mostrar o e-mail de acesso -- ------------------------------------------------------------ create or replace function admin_vendedores() returns json language plpgsql security definer set search_path = public as $$ declare v_out json; begin if not is_admin() then raise exception 'Acesso restrito.'; end if; select coalesce(json_agg(x order by x.nome), '[]'::json) into v_out from ( select ve.id, ve.nome, ve.telefone, ve.codigo, ve.comissao_pct, ve.ativo, ve.user_id, ve.criado_em, (select u.email from auth.users u where u.id = ve.user_id) as email, coalesce((select count(*) from vendas v where v.vendedor_id = ve.id and v.status = 'pago'), 0) as vendas_pagas, coalesce((select sum(v.valor_centavos) from vendas v where v.vendedor_id = ve.id and v.status = 'pago'), 0) as total_vendido, coalesce((select sum(v.comissao_centavos) from vendas v where v.vendedor_id = ve.id and v.status = 'pago'), 0) as total_comissao from vendedores ve ) x; return v_out; end $$;
